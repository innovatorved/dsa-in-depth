/**
 * Computes the Z-Array z[0..N-1] in exact Theta(N) time and space.
 * z[i] = length of the longest common prefix between S and the suffix S[i..N-1].
 */
function computeZArray(str) {
  const n = str.length;
  const z = new Int32Array(n);
  if (n === 0) return z;
  z[0] = n;

  let l = 0, r = 0;
  for (let i = 1; i < n; i++) {
    if (i <= r) {
      const k = i - l;
      const rem = r - i + 1;
      z[i] = z[k] < rem ? z[k] : rem;
    }
    while (i + z[i] < n && str.charCodeAt(z[i]) === str.charCodeAt(i + z[i])) {
      z[i]++;
    }
    if (i + z[i] - 1 > r) {
      l = i;
      r = i + z[i] - 1;
    }
  }
  return z;
}

/**
 * Finds all occurrences of <code>pattern</code> in <code>text</code> using the Z-Algorithm without
 * allocating a concatenated string P + '$' + T! Uses only O(M) extra memory.
 */
function zPatternSearch(text, pattern) {
  const n = text.length, m = pattern.length;
  if (m === 0) return [];
  const zPat = computeZArray(pattern);
  const matches = [];

  let l = -1, r = -1;
  for (let i = 0; i < n; i++) {
    let matchLen = 0;
    if (i <= r) {
      const k = i - l;
      const rem = r - i + 1;
      matchLen = zPat[k] < rem ? zPat[k] : rem;
    }
    while (
      matchLen < m &&
      i + matchLen < n &&
      pattern.charCodeAt(matchLen) === text.charCodeAt(i + matchLen)
    ) {
      matchLen++;
    }
    if (i + matchLen - 1 > r) {
      l = i;
      r = i + matchLen - 1;
    }
    if (matchLen === m) {
      matches.push(i);
    }
  }
  return matches;
}

std::vector<int> compute_z_array(std::string_view s) {
    int n = static_cast<int>(s.size());
    std::vector<int> z(n, 0);
    if (n == 0) return z;
    z[0] = n;
    for (int i = 1, l = 0, r = 0; i < n; ++i) {
        if (i <= r) z[i] = std::min(r - i + 1, z[i - l]);
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) ++z[i];
        if (i + z[i] - 1 > r) {
            l = i;
            r = i + z[i] - 1;
        }
    }
    return z;
}

std::vector<int> z_pattern_search(std::string_view text, std::string_view pattern) {
    int n = static_cast<int>(text.size()), m = static_cast<int>(pattern.size());
    if (m == 0) return {};
    std::vector<int> z_pat = compute_z_array(pattern);
    std::vector<int> matches;
    for (int i = 0, l = -1, r = -1; i < n; ++i) {
        int match_len = (i <= r) ? std::min(r - i + 1, z_pat[i - l]) : 0;
        while (match_len < m && i + match_len < n && pattern[match_len] == text[i + match_len]) {
            ++match_len;
        }
        if (i + match_len - 1 > r) {
            l = i;
            r = i + match_len - 1;
        }
        if (match_len == m) matches.push_back(i);
    }
    return matches;
}

Key Takeaways

The Z-array entry Z[i] is the length of the longest common prefix between S and the suffix S[i..N-1].
Maintaining the rightmost Z-box [L, R] where S[L..R] = S[0..R-L] allows initializing Z[i] = min(R - i + 1, Z[i - L]) whenever i <= R.
Every character comparison that succeeds inside the while loop advances R by +1, proving that the Z-algorithm performs at most 2N character comparisons in the worst case.
Precomputing Z[] on the pattern alone allows streaming Z-box matching across text T in Theta(N + M) time and Theta(M) memory without a separator character.
