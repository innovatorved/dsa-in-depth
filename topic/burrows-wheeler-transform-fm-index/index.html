Intuition first

How does bzip2 permute a text so that identical characters cluster into long runs without losing a single bit of information, and how do genomics aligners (Bowtie2, BWA) index the 3-billion-base-pair human genome in under 2 GB of RAM while still counting exact occurrences of a length-MMM DNA read in O(M)O(M)O(M) time? Both feats rely on the Burrows-Wheeler Transform (BWT, 1994) and Paolo Ferragina and Giovanni Manzini's FM-Index (2000).

#include <array>
#include <cstdint>
#include <string>
#include <string_view>
#include <utility>
#include <vector>

class FMIndex {
public:
    // Builds BWT and FM-Index from string <code>s</code> (terminated by '' at s.back()) and its SA
    FMIndex(std::string_view s_with_null, const std::vector<int>& sa)
        : n_(static_cast<int>(s_with_null.size())), bwt_(n_, ''), occ_(n_ + 1) {
        c_.fill(0);
        occ_[0].fill(0);
        for (int r = 0; r < n_; ++r) {
            int prev_idx = (sa[r] == 0) ? (n_ - 1) : (sa[r] - 1);
            uint8_t ch = static_cast<uint8_t>(s_with_null[prev_idx]);
            bwt_[r] = static_cast<char>(ch);
            c_[ch]++;
            occ_[r + 1] = occ_[r];
            occ_[r + 1][ch]++;
        }
        // Convert c_ into exclusive prefix sums (bucket start offsets in F)
        int sum = 0;
        for (int ch = 0; ch < 256; ++ch) {
            int cnt = c_[ch];
            c_[ch] = sum;
            sum += cnt;
        }
    }

    // Reconstructs the original string (excluding trailing '') in O(N) via LF-mapping
    std::string invert_bwt() const {
        std::string out(n_ - 1, '');
        int r = 0; // Row 0 in F starts with '', so L[0] is the last char of S
        for (int i = n_ - 2; i >= 0; --i) {
            uint8_t ch = static_cast<uint8_t>(bwt_[r]);
            out[i] = static_cast<char>(ch);
            r = c_[ch] + occ_[r][ch];
        }
        return out;
    }

    // Counts exact occurrences of <code>pat</code> in O(|pat|) time via FM-Index Backward Search
    int count_matches(std::string_view pat) const {
        int sp = 0, ep = n_;
        for (int j = static_cast<int>(pat.size()) - 1; j >= 0 && sp < ep; --j) {
            uint8_t ch = static_cast<uint8_t>(pat[j]);
            sp = c_[ch] + occ_[sp][ch];
            ep = c_[ch] + occ_[ep][ch];
        }
        return std::max(0, ep - sp);
    }

private:
    int n_;
    std::string bwt_;
    std::array<int, 256> c_{};
    std::vector<std::array<int, 256>> occ_;
};

Key Takeaways

The Burrows-Wheeler Transform L[r] = S[(SA[r] - 1 + N) mod N] groups characters that precede identical contexts into long homogeneous runs ideal for Move-to-Front and Run-Length compression.
The Last-to-First (LF) property guarantees that the k-th occurrence of character c in L is the exact same text character as the k-th occurrence of c in F, enabling O(N) inversion via LF(r) = C[L[r]] + Occ(L[r], r).
FM-Index backward search processes pattern P from right to left (j = M - 1 down to 0) updating [sp, ep) = [C[c] + Occ(c, sp), C[c] + Occ(c, ep)) to count occurrences in O(M) time.
