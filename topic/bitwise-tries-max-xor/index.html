Shows how inserting fixed-width integers from MSB to LSB into a binary trie turns maximum XOR queries into a greedy O(W) bit-flip walk.

Intuition first

When keys are fixed-width WWW-bit integers (e.g., W=30W = 30W=30 or 646464), viewing each integer as a binary string from Most Significant Bit (MSB, bit W−1W-1W−1) down to Least Significant Bit (LSB, bit 000) turns a Prefix Trie into a Bitwise (0-1) Trie of fixed depth WWW. Because 2k>∑j=0k−12j2^{k} > \sum_{j=0}^{k-1} 2^{j}2k>∑j=0k−1​2j, gaining a 111 at bit position kkk is strictly more valuable than gaining 111s at all lower bit positions combined!

This greedy dominance makes Bitwise Tries the natural engine for XOR problems: to maximize x⊕yx \oplus yx⊕y for a query xxx over a set SSS, walk down the trie from bit W−1W-1W−1 to 000, and at each bit b=(x≫i)&1b = (x \gg i) \mathrel{\&} 1b=(x≫i)&1, greedily take the opposite child b⊕1b \oplus 1b⊕1 if it exists (setting bit iii of the XOR result to 111), or fall back to child bbb otherwise. Combining this O(W)O(W)O(W) query with the prefix-XOR identity XOR(L..R)=P[R]⊕P[L−1]\text{XOR}(L..R) = P[R] \oplus P[L-1]XOR(L..R)=P[R]⊕P[L−1] solves Maximum XOR Subarray in O(N⋅W)O(N \cdot W)O(N⋅W) time; adding a + "cnt" + field supports dynamic deletions; and storing the maximum insertion index + "maxIdx[u]" + at each trie node answers offline subarray XOR queries [L,R][L, R][L,R] without even needing persistence!

1. MSB-to-LSB Greedy Dominance & Dynamic Deletion via cnt[u]

Why must a Bitwise Trie for XOR maximization or order-statistic comparison be rooted at the Most Significant Bit (W−1W - 1W−1) rather than bit 000? Because binary value order is lexicographical from MSB to LSB: every number in the 1-branch at depth kkk is strictly greater than every number in the 0-branch, since 2k>2k−1=∑j=0k−12j2^k > 2^k - 1 = \sum_{j=0}^{k-1} 2^j2k>2k−1=∑j=0k−1​2j.

To support dynamic insert(x) and remove(x) (such as sliding-window XOR or multiset queries), we store an integer counter cnt[u] at every trie node recording how many active numbers pass through node uuu. Instead of physically freeing nodes on remove(x), we simply decrement cnt[u]-- along xxx's path, and treat any child ccc with c === 0 || cnt[c] === 0 as empty!

/**
 * High-Performance 30-Bit Binary Trie using a Flat TypedArray Pool
 * Supports dynamic insert, remove, maxXor(x), and countXorLess(x, K) in O(W) time.
 */
class BitwiseTrie {
  constructor(maxInsertions = 100005, bitWidth = 30) {
    this.W = bitWidth;
    const maxNodes = maxInsertions * this.W + 2;
    this.ch = new Int32Array(maxNodes * 2); // ch[(u << 1) | bit]
    this.cnt = new Int32Array(maxNodes);
    this.ptr = 1; // Node 1 is root; Node 0 is null sentinel
  }

  insert(x, delta = 1) {
    let u = 1;
    this.cnt[u] += delta;
    for (let i = this.W - 1; i >= 0; i--) {
      const b = (x >>> i) & 1;
      const edgeIdx = (u << 1) | b;
      let next = this.ch[edgeIdx];
      if (next === 0) {
        next = ++this.ptr;
        this.ch[edgeIdx] = next;
      }
      u = next;
      this.cnt[u] += delta;
    }
  }

  remove(x) {
    this.insert(x, -1);
  }

  maxXor(x) {
    let u = 1;
    if (this.cnt[u] === 0) return 0;
    let ans = 0;
    for (let i = this.W - 1; i >= 0; i--) {
      const b = (x >>> i) & 1;
      const want = b ^ 1;
      const desiredChild = this.ch[(u << 1) | want];
      if (desiredChild !== 0 && this.cnt[desiredChild] > 0) {
        ans |= (1 << i);
        u = desiredChild;
      } else {
        u = this.ch[(u << 1) | b];
      }
    }
    return ans >>> 0;
  }

  /**
   * Returns count of active elements y in the trie such that (x ^ y) < k
   */
  countXorLess(x, k) {
    let u = 1;
    let total = 0;
    for (let i = this.W - 1; i >= 0 && u !== 0; i--) {
      const bx = (x >>> i) & 1;
      const bk = (k >>> i) & 1;
      if (bk === 1) {
        const sameChild = this.ch[(u << 1) | bx];
        if (sameChild !== 0) total += this.cnt[sameChild];
        u = this.ch[(u << 1) | (bx ^ 1)];
      } else {
        u = this.ch[(u << 1) | bx];
      }
    }
    return total;
  }
}

/**
 * Maximum XOR of any contiguous subarray A[L..R] in O(N * W) time
 */
function maxXorSubarray(arr, bitWidth = 30) {
  const trie = new BitwiseTrie(arr.length + 1, bitWidth);
  trie.insert(0); // Prefix XOR P[0] = 0 so subarrays starting at index 0 are included!
  let pref = 0;
  let best = 0;
  for (let i = 0; i < arr.length; i++) {
    pref ^= arr[i];
    best = Math.max(best, trie.maxXor(pref));
    trie.insert(pref);
  }
  return best;
}

// Flat-Pool 30-Bit Binary Trie for O(W) Max-XOR & Count-XOR-Less-Than-K (C++20)
class BitwiseTrie {
    static constexpr int W = 30;
    std::vector<std::array<int, 2>> ch;
    std::vector<int> cnt;
public:
    explicit BitwiseTrie(int max_insertions = 100005) {
        ch.reserve(max_insertions * W + 2);
        cnt.reserve(max_insertions * W + 2);
        ch.push_back({0, 0}); cnt.push_back(0); // 0 = null sentinel
        ch.push_back({0, 0}); cnt.push_back(0); // 1 = root
    }

    void insert(uint32_t x, int delta = 1) {
        int u = 1;
        cnt[u] += delta;
        for (int i = W - 1; i >= 0; --i) {
            int b = (x >> i) & 1;
            if (ch[u][b] == 0) {
                ch[u][b] = static_cast<int>(ch.size());
                ch.push_back({0, 0});
                cnt.push_back(0);
            }
            u = ch[u][b];
            cnt[u] += delta;
        }
    }

    uint32_t max_xor(uint32_t x) const noexcept {
        int u = 1;
        if (cnt[u] == 0) return 0;
        uint32_t ans = 0;
        for (int i = W - 1; i >= 0; --i) {
            int b = (x >> i) & 1;
            int want = b ^ 1;
            int nxt = ch[u][want];
            if (nxt != 0 && cnt[nxt] > 0) {
                ans |= (1u << i);
                u = nxt;
            } else {
                u = ch[u][b];
            }
        }
        return ans;
    }

    int count_xor_less(uint32_t x, uint32_t k) const noexcept {
        int u = 1, total = 0;
        for (int i = W - 1; i >= 0 && u != 0; --i) {
            int bx = (x >> i) & 1;
            int bk = (k >> i) & 1;
            if (bk == 1) {
                int same = ch[u][bx];
                if (same != 0) total += cnt[same];
                u = ch[u][bx ^ 1];
            } else {
                u = ch[u][bx];
            }
        }
        return total;
    }
};

Key Takeaways

A Bitwise (0-1) Trie inserts W-bit integers from MSB (bit W-1) down to LSB (bit 0) because 2^k exceeds the sum of all lower powers of two.
Maximizing x ^ y greedily chooses child (b ^ 1) at each bit i if available (cnt > 0), falling back to child b in O(W) time.
Counting y with (x ^ y) < K adds cnt[child[bx]] whenever bit i of K is 1, then descends into child[bx ^ 1] where the prefix of (x ^ y) matches K.
Subarray XOR problems reduce to Pair XOR on the prefix array P[i] = P[i-1] ^ A[i], provided P[0] = 0 is inserted into the trie first.
