Explains the polynomial view of strings and how sliding a window updates the hash in O(1) arithmetic operations.

Intuition first

Karp and Rabin (1987) turned string comparison into modular arithmetic by treating a string S[0..N−1]S[0..N-1]S[0..N−1] as a polynomial evaluated at a random base BBB modulo a prime MMM: H(S)=(∑i=0N−1S[i]⋅BN−1−i) mod MH(S) = \left(\sum_{i=0}^{N-1} S[i] \cdot B^{N-1-i}\right) \bmod MH(S)=(∑i=0N−1​S[i]⋅BN−1−i)modM. By precomputing prefix hashes P[i]=(P[i−1]⋅B+S[i]) mod MP[i] = (P[i-1] \cdot B + S[i]) \bmod MP[i]=(P[i−1]⋅B+S[i])modM and base powers Bk mod MB^k \bmod MBkmodM in O(N)O(N)O(N) time, the hash of any arbitrary substring S[L..R]S[L..R]S[L..R] can be extracted in O(1)O(1)O(1) time via (P[R+1]−P[L]⋅BR−L+1) mod M(P[R+1] - P[L] \cdot B^{R - L + 1}) \bmod M(P[R+1]−P[L]⋅BR−L+1)modM! This unlocks algorithms that deterministic automata cannot easily express, such as O(log⁡N)O(\log N)O(logN) Longest Common Prefix (LCP) of any two substrings via binary search, or O(Nlog⁡N)O(N \log N)O(NlogN) Longest Duplicate Substring.

However, naive 32-bit rolling hashes ( + "M = 10^9 + 7" + ) fail catastrophically whenever a problem compares Θ(N)\Theta(N)Θ(N) substring hashes: by the Birthday Paradox, comparing K=105K = 10^5K=105 hashes has collision probability ≈1−e−K2/(2M)≈99.3%\approx 1 - e^{-K^2 / (2M)} \approx 99.3\%≈1−e−K2/(2M)≈99.3%! 2. Production Implementation: Mersenne 261−12^{61}-1261−1 Rolling Hash with O(1)O(1)O(1) Substring Hash & O(log⁡N)O(\log N)O(logN) LCP

const MOD61 = (1n << 61n) - 1n;
// Random odd base chosen per process startup in [257 .. 10^9] to defeat adversarial inputs
const RANDOM_BASE = BigInt(257 + Math.floor(Math.random() * 1000000000));

function mod61(x) {
  let res = (x & MOD61) + (x >> 61n);
  if (res >= MOD61) res -= MOD61;
  return res;
}

function mul61(a, b) {
  return mod61(a * b);
}

/**
 * RollingHash61: O(N) build, O(1) arbitrary substring hash, O(log N) substring LCP & compare
 */
class RollingHash61 {
  constructor(str, base = RANDOM_BASE) {
    this.str = str;
    this.n = str.length;
    this.pref = new BigUint64Array(this.n + 1);
    this.pow = new BigUint64Array(this.n + 1);
    this.pow[0] = 1n;

    for (let i = 0; i < this.n; i++) {
      const code = BigInt(str.charCodeAt(i));
      this.pref[i + 1] = mod61(this.pref[i] * base + code);
      this.pow[i + 1] = mul61(this.pow[i], base);
    }
  }

  // O(1) hash of closed substring str[l..r] (0-indexed); returns 0n for empty range l > r
  getHash(l, r) {
    if (l > r) return 0n;
    const sub = this.pref[r + 1] + MOD61 - mul61(this.pref[l], this.pow[r - l + 1]);
    return sub >= MOD61 ? sub - MOD61 : sub;
  }

  // O(log N) Longest Common Prefix of suffixes str[i..] and str[j..]
  lcp(i, j) {
    let lo = 0;
    let hi = this.n - Math.max(i, j);
    while (lo < hi) {
      const mid = (lo + hi + 1) >> 1;
      if (this.getHash(i, i + mid - 1) === this.getHash(j, j + mid - 1)) {
        lo = mid;
      } else {
        hi = mid - 1;
      }
    }
    return lo;
  }

  // O(log N) lexicographical comparison of substrings str[l1..r1] and str[l2..r2]
  compareSubstrings(l1, r1, l2, r2) {
    const len1 = r1 - l1 + 1, len2 = r2 - l2 + 1;
    const common = Math.min(len1, len2, this.lcp(l1, l2));
    if (common >= Math.min(len1, len2)) return len1 - len2;

// Mersenne Prime 2^61 - 1 Polynomial Rolling Hash with __uint128_t (C++20)
class RollingHash61 {
    static constexpr uint64_t MOD = (1ULL << 61) - 1;
    std::string s;
    std::vector<uint64_t> pref, pw;

    static constexpr uint64_t mod61(__uint128_t x) noexcept {
        uint64_t res = static_cast<uint64_t>(x & MOD) + static_cast<uint64_t>(x >> 61);
        return res >= MOD ? res - MOD : res;
    }
    static constexpr uint64_t mul61(uint64_t a, uint64_t b) noexcept {
        return mod61(static_cast<__uint128_t>(a) * b);
    }
public:
    explicit RollingHash61(std::string_view str, uint64_t base = 911382323ULL)
        : s(str), pref(str.size() + 1, 0), pw(str.size() + 1, 1) {
        for (size_t i = 0; i < s.size(); ++i) {
            pref[i + 1] = mod61(static_cast<__uint128_t>(pref[i]) * base + static_cast<unsigned char>(s[i]));
            pw[i + 1] = mul61(pw[i], base);
        }
    }

    uint64_t get_hash(int l, int r) const noexcept {
        if (l > r) return 0;
        uint64_t sub = pref[r + 1] + MOD - mul61(pref[l], pw[r - l + 1]);
        return sub >= MOD ? sub - MOD : sub;
    }

    int lcp(int i, int j) const noexcept {
        int lo = 0, hi = static_cast<int>(s.size()) - std::max(i, j);
        while (lo < hi) {
            int mid = (lo + hi + 1) >> 1;
            if (get_hash(i, i + mid - 1) == get_hash(j, j + mid - 1)) lo = mid;
            else hi = mid - 1;
        }
        return lo;
    }

Key Takeaways

Precomputing prefix polynomial hashes P[i] = (P[i-1]*B + S[i-1]) mod M allows O(1) hash extraction of any substring S[L..R] via (P[R+1] - P[L]*B^(R-L+1)) mod M.
By the Birthday Paradox, comparing K = 10^5 substrings requires a modulus M >> K^2; a single 32-bit prime (10^9 + 7) collides with >99% probability.
The Mersenne prime M = 2^61 - 1 with a runtime-randomized base supports division-free reduction (x & M) + (x >> 61) and defeats both Birthday collisions and Thue-Morse anti-hash strings.
Combining O(1) substring hashing with binary search computes the Longest Common Prefix (LCP) and lexicographical order of any two substrings in O(log N) time.
